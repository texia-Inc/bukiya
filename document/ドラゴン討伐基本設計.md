## 3. 討伐システムの詳細設計

### 3.1 戦闘シミュレーション

#### 自動戦闘の計算方式
```python
class DragonRaidSimulation:
    def __init__(self, dragon, participants):
        self.dragon = dragon
        self.participants = participants
        self.battle_phases = []
        
    def simulate_battle(self):
        """戦闘全体をシミュレーション"""
        
        # Phase 1: 初期配置・戦略確認
        phase1_result = self.simulate_phase1()
        
        # Phase 2: 主力戦闘
        phase2_result = self.simulate_phase2(phase1_result)
        
        # Phase 3: 最終決戦
        phase3_result = self.simulate_phase3(phase2_result)
        
        return self.compile_results()
    
    def calculate_damage_per_turn(self):
        """1ターンあたりのダメージ計算"""
        total_damage = 0
        
        for participant in self.participants:
            # 基本攻撃力
            base_damage = participant.weapon.attack_power
            
            # 属性相性ボーナス
            elemental_bonus = self.get_elemental_bonus(
                participant.weapon.element, 
                self.dragon.weakness
            )
            
            # 冒険者スキルボーナス
            skill_bonus = participant.get_skill_bonus(self.dragon.type)
            
            # 信頼度ボーナス
            trust_bonus = participant.trust_level * 0.01
            
            final_damage = base_damage * elemental_bonus * skill_bonus * (1 + trust_bonus)
            total_damage += final_damage
            
        return total_damage
```

class DragonRaidProgress:
    def __init__(self):
        self.last_update = time.time()
        self.update_interval = 30  # 30秒間隔
        
    def get_progress_update(self):
        """軽量な差分更新のみ送信"""
        current_time = time.time()
        if current_time - self.last_update >= self.update_interval:
            return {
                'dragon_hp_percent': self.calculate_hp_percent(),
                'phase_progress': self.get_phase_progress(),
                'latest_events': self.get_recent_events(30),  # 30秒分のみ
                'participant_count': self.get_participant_count()
            }
        return None
