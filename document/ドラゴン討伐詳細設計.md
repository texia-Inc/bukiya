### 2.1 基本ダメージ計算式

def calculate_damage(adventurer, dragon, weapon):
    # 基本攻撃力
    base_attack = weapon.attack_power + (weapon.enchant_level * weapon.enchant_rate)
    
    # 冒険者レベル補正
    level_bonus = 1.0 + (adventurer.level * 0.02)  # レベル1あたり2%
    
    # 武器種補正
    weapon_type_bonus = get_weapon_type_bonus(weapon.type, dragon.type)
    
    # 属性相性
    elemental_multiplier = get_elemental_bonus(weapon.element, dragon.weakness)
    
    # 信頼度ボーナス
    trust_bonus = 1.0 + (adventurer.trust_level * 0.003)  # 信頼度1あたり0.3%
    
    # クリティカル判定
    crit_chance = 0.05 + (adventurer.luck * 0.001)  # 基本5% + 運補正
    is_critical = random.random() < crit_chance
    critical_multiplier = 2.0 if is_critical else 1.0
    
    # 最終ダメージ計算
    final_damage = (
        base_attack * 
        level_bonus * 
        weapon_type_bonus * 
        elemental_multiplier * 
        trust_bonus * 
        critical_multiplier
    )
    
    # ランダム要素 (±10%)
    random_factor = random.uniform(0.9, 1.1)
    
    return int(final_damage * random_factor), is_critical

### 2.2 スキル・特殊効果システム
```python
class AdventurerSkills:
    def __init__(self, adventurer):
        self.adventurer = adventurer
        self.skills = self.determine_skills()
    
    def determine_skills(self):
        """冒険者のレベルと職業に基づいてスキル決定"""
        skills = []
        
        # レベルベースのスキル
        if self.adventurer.level >= 10:
            skills.append('PowerStrike')  # 強力な一撃
        if self.adventurer.level >= 20:
            skills.append('Concentration')  # 集中力向上
        if self.adventurer.level >= 30:
            skills.append('BattleFrenzy')  # 戦闘狂化
            
        # 職業別スキル
        if self.adventurer.class_type == 'warrior':
            skills.extend(['ShieldBash', 'Charge', 'LastStand'])
        elif self.adventurer.class_type == 'mage':
            skills.extend(['ElementalBoost', 'ManaShield', 'Meditation'])
        elif self.adventurer.class_type == 'archer':
            skills.extend(['PrecisionShot', 'MultiShot', 'Eagle Eye'])
            
        return skills
    
    def activate_skill(self, battle_context):
        """戦闘中のスキル発動判定"""
        for skill in self.skills:
            if self.should_activate_skill(skill, battle_context):
                return self.execute_skill(skill, battle_context)
        return None
    
    def should_activate_skill(self, skill, context):
        """スキル発動条件の判定"""
        conditions = {
            'PowerStrike': context.dragon_hp_percent < 0.5,  # ドラゴンHP50%以下
            'LastStand': context.adventurer_hp_percent < 0.3,  # 自分のHP30%以下
            'ElementalBoost': context.has_matching_element,  # 属性一致時
            'PrecisionShot': context.turn_count % 5 == 0,  # 5ターンに1回
        }
        
        base_chance = 0.1  # 基本10%の発動率
        if skill in conditions and conditions[skill]:
            base_chance *= 2  # 条件満たす時は発動率2倍
            
        return random.random() < base_chance

class BattleTextGenerator:

            ]
        }
        
        self.reaction_texts = {
            'high_damage': [
                '→ {target}に{damage}の大ダメージ！',
                '→ {damage}ダメージ！{target}が大きく怯んだ！',
                '→ 会心の{damage}ダメージ！'
            ],
            'normal_damage': [
                '→ {target}に{damage}ダメージ',
                '→ {damage}ダメージを与えた！',
                '→ {target}のHPが{damage}減少！'
            ],
            'low_damage': [
                '→ {target}に{damage}ダメージ... 効果はいまひとつ',
                '→ {damage}ダメージ。{target}は堅い！',
                '→ わずか{damage}ダメージ...'
            ]
        }
    
    def generate_attack_text(self, attacker, target, damage, is_critical, 
                           elemental_bonus, skill_used):
        """攻撃テキストの生成"""
        texts = []
        
        # 基本攻撃テキスト
        if skill_used:
            base_text = random.choice(self.damage_texts['skill'])
            texts.append(base_text.format(
                adventurer=attacker.name,
                skill_name=skill_used.name
            ))
        elif is_critical:
            base_text = random.choice(self.damage_texts['critical'])
            texts.append(base_text.format(
                adventurer=attacker.name,
                target=target.name
            ))
        elif elemental_bonus > 1.0:
            base_text = random.choice(self.damage_texts['elemental'])
            texts.append(base_text.format(
                adventurer=attacker.name,
                weapon=attacker.weapon.name,
                element=attacker.weapon.element
            ))
        else:
            base_text = random.choice(self.damage_texts['normal'])
            texts.append(base_text.format(
                adventurer=attacker.name,
                target=target.name,
                weapon=attacker.weapon.name
            ))
        
        # ダメージ結果テキスト
        damage_level = self.categorize_damage(damage, target.max_hp)
        reaction_text = random.choice(self.reaction_texts[damage_level])
        texts.append(reaction_text.format(
            target=target.name,
            damage=f"{damage:,}"
        ))
        
        # 特殊効果の追加
        if is_critical:
            texts.append('💥 クリティカルヒット！')
        if elemental_bonus > 1.0:
            texts.append(f'⚡ 属性相性で威力{elemental_bonus:.1f}倍！')
        if skill_used:
            texts.append(f'{skill_used.effect} {skill_used.description}')
            
        return texts
    
    def categorize_damage(self, damage, target_max_hp):
        """ダメージレベルの分類"""
        damage_ratio = damage / target_max_hp
        if damage_ratio > 0.005:  # 最大HPの0.5%以上
            return 'high_damage'
        elif damage_ratio > 0.001:  # 最大HPの0.1%以上
            return 'normal_damage'
        else:
            return 'low_damage'
```

            '⚡ 支援魔法が発動！攻撃力が一時的に上昇！'
        ]
    
    def check_special_events(self, battle_state):
        """特殊イベントの発生判定"""
        events = []
        
        # HP閾値イベント
        current_hp_ratio = battle_state.dragon.current_hp / battle_state.dragon.max_hp
        for threshold, event_texts in self.hp_threshold_events.items():
            if (battle_state.last_hp_ratio > threshold and 
                current_hp_ratio <= threshold):
                events.append(random.choice(event_texts))
                battle_state.trigger_special_phase(threshold)
        
        # コンボ攻撃イベント
        if battle_state.consecutive_hits >= 5:
            events.append(random.choice(self.combo_texts))
            battle_state.apply_combo_bonus()
        
        # 支援イベント
        if battle_state.needs_support():
            events.append(random.choice(self.support_texts))
            battle_state.apply_support_effect()
            
        return events

## 4. 自動進行システム

### 4.1 戦闘ターン管理
```python
class AutoBattleManager:
    def __init__(self, participants, dragon):
        self.participants = participants
        self.dragon = dragon
        self.turn_count = 0
        self.battle_log = []
        self.update_interval = 10  # 10秒ごとに更新
        
    def run_battle_turn(self):
        """1ターンの戦闘処理"""
        turn_events = []
        
        # 各参加者の攻撃処理
        for participant in self.participants:
            if participant.is_alive and self.dragon.is_alive:
                attack_result = self.process_attack(participant, self.dragon)
                turn_events.extend(attack_result.events)
                
                # 特殊イベントの判定
                special_events = self.check_special_events()
                turn_events.extend(special_events)
        
        # ドラゴンの反撃
        if self.dragon.is_alive:
            dragon_attack = self.process_dragon_attack()
            turn_events.extend(dragon_attack.events)
        
        # ターン終了処理
        self.turn_count += 1
        self.battle_log.extend(turn_events)
        
        return BattleTurnResult(
            events=turn_events,
            dragon_hp=self.dragon.current_hp,
            participants_alive=len([p for p in self.participants if p.is_alive]),
            turn_number=self.turn_count
        )
    
    def process_attack(self, attacker, target):
        """個別の攻撃処理"""
        # ダメージ計算
        damage, is_critical = calculate_damage(attacker, target, attacker.weapon)
        
        # スキル判定
        skill_used = attacker.try_activate_skill()
        if skill_used:
            damage = int(damage * skill_used.multiplier)
        
        # ダメージ適用
        actual_damage = target.take_damage(damage)
        
        # テキスト生成
        attack_texts = self.text_generator.generate_attack_text(
            attacker, target, actual_damage, is_critical, 
            self.get_elemental_bonus(attacker, target), skill_used
        )
        
        return AttackResult(
            damage=actual_damage,
            is_critical=is_critical,
            skill_used=skill_used,
            events=attack_texts
        )
```

### 4.2 リアルタイム更新
```python
class BattleUpdater:
    def __init__(self, battle_manager):
        self.battle_manager = battle_manager
        self.last_update_time = time.time()
        self.update_queue = []
        
    async def auto_update_loop(self):
        """自動更新ループ"""
        while self.battle_manager.is_active:
            current_time = time.time()
            
            if current_time - self.last_update_time >= 10:  # 10秒間隔
                # 戦闘ターン実行
                turn_result = self.battle_manager.run_battle_turn()
                
                # クライアントに送信するデータ準備
                update_data = {
                    'timestamp': current_time,
                    'dragon_hp': turn_result.dragon_hp,
                    'dragon_hp_percent': turn_result.dragon_hp / self.battle_manager.dragon.max_hp,
                    'participants_alive': turn_result.participants_alive,
                    'new_events': turn_result.events[-5:],  # 最新5件のイベント
                    'turn_number': turn_result.turn_number,
                    'phase': self.battle_manager.current_phase
                }
                
                # 全クライアントに配信
                await self.broadcast_update(update_data)
                
                self.last_update_time = current_time
            
            await asyncio.sleep(1)  # 1秒待機
    
    async def broadcast_update(self, data):
        """全参加者に更新データを配信"""
        # WebSocket経由でクライアントに送信
        for client in self.connected_clients:
            try:
                await client.send_json(data)
            except:
                # 接続が切れたクライアントは除外
                self.connected_clients.remove(client)
```

dartclass DragonBattleScreen extends StatefulWidget {

  _DragonBattleScreenState createState() => _DragonBattleScreenState();

class _DragonBattleScreenState extends State<DragonBattleScreen> {
  Timer? _updateTimer;
  List<BattleLogEntry> _battleLog = [];
  double _dragonHpPercent = 1.0;
  
  @override
  void initState() {
    super.initState();
    _startAutoUpdate();
  }
  
  void _startAutoUpdate() {
    _updateTimer = Timer.periodic(Duration(seconds: 10), (timer) {
      _fetchBattleUpdate();
    });
  }
  
  void _fetchBattleUpdate() async {
    final update = await BattleAPI.getBattleUpdate();
    setState(() {
      _dragonHpPercent = update.dragonHpPercent;
      _battleLog.addAll(update.newEvents);
      
      // ログが長すぎる場合は古いものを削除
      if (_battleLog.length > 50) {
        _battleLog = _battleLog.takeLast(50).toList();
      }
    });
    
    // ログを最新に自動スクロール
    _scrollToBottom();
  }
}
